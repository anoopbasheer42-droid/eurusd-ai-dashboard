function getBestVoice() { const voices = window.speechSynthesis.getVoices(); if (!voices || voices.length === 0) return null; // Wait for voices to load const priorities = ['Google UK English Female', 'Google US English', 'Microsoft Zira', 'Microsoft Aria', 'Samantha', 'Karen', 'Moira', 'Tessa', 'Victoria', 'Google UK English Male', 'Daniel', 'Alex']; for (const name of priorities) { const found = voices.find(v => v.name.includes(name)); if (found) return found; } return voices.find(v => v.lang.startsWith('en')) || voices[0]; } function fallbackSpeakText(text, isReminder = false) { if (!('speechSynthesis' in window)) return; window.speechSynthesis.cancel(); // If voices aren't loaded yet, wait 500ms and try again let voice = getBestVoice(); if (!voice) { setTimeout(() => fallbackSpeakText(text, isReminder), 500); return; } const sentences = text.split(/(?<=[.!?,])\s+/).map(s => s.trim()).filter(s => s.length > 0); if (sentences.length === 0) return; isSpeaking = true; let index = 0; function speakNext() { if (index >= sentences.length) { isSpeaking = false; return; } const sentence = sentences[index]; const u = new SpeechSynthesisUtterance(sentence); u.voice = voice; const basePitch = isReminder ? 1.05 : 1.1; const baseRate = isReminder ? 0.95 : 1.0; u.pitch = Math.max(0.8, Math.min(1.5, basePitch + (Math.random() - 0.5) * 0.05)); u.rate = Math.max(0.8, Math.min(1.3, baseRate + (Math.random() - 0.5) * 0.05)); u.volume = 1.0; u.onend = () => { index++; setTimeout(speakNext, 350 + Math.random() * 200); }; u.onerror = (e) => { console.error("Speech error:", e); isSpeaking = false; }; window.speechSynthesis.speak(u); } speakNext(); }
